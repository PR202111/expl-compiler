decl
    int n, arr[50], i, j, temp;
    str msg;
enddecl
begin
    msg = "Enter N:";
    write(msg);
    read(n);
    
    msg = "Enter Elements:";
    write(msg);
    i = 0;
    while(i < n) do
        read(arr[i]);
        i = i + 1;
    endwhile;

    i = 0;
    while(i < n) do
        j = 0;
        while(j < (n - 1 - i)) do
            if(arr[j] > arr[j+1]) then
                temp = arr[j];
                arr[j] = arr[j+1];
                arr[j+1] = temp;
            endif;
            j = j + 1;
        endwhile;
        i = i + 1;
    endwhile;

    msg = "Sorted Array:";
    write(msg);
    i = 0;
    while(i < n) do
        write(arr[i]);
        i = i + 1;
    endwhile;
end;
